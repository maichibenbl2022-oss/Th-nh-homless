local function setupM2DynamicListener(p)
    if p == LocalPlayer then return end
    local function onChar(char)
        if not char then return end
        local hum = char:WaitForChild("Humanoid", 5)
        local root = char:WaitForChild("HumanoidRootPart", 5)
        if not hum or not root then return end
        
        hum.AnimationPlayed:Connect(function(track)
            if not isEnabled then return end
            local animId = track.Animation and track.Animation.AnimationId
            if animId == M2_TargetID then
                local myChar = LocalPlayer.Character
                if myChar and myChar:FindFirstChild("HumanoidRootPart") then
                    local myPos = myChar.HumanoidRootPart.Position
                    local cPos = root.Position
                    local dx = cPos.X - myPos.X
                    local dy = cPos.Y - myPos.Y
                    local dz = cPos.Z - myPos.Z
                    if (dx * dx + dy * dy + dz * dz) <= COMBAT_RANGE_SQ then
                        activeM2Count = activeM2Count + 1
                        rawUnblock()
                        local cleaned = false
                        local stoppedConn, endedConn, diedConn
                        local timeoutTask
                        local function cleanup()
                            if cleaned then return end
                            cleaned = true
                            activeM2Count = math.max(0, activeM2Count - 1)
                            if stoppedConn then stoppedConn:Disconnect() end
                            if endedConn then endedConn:Disconnect() end
                            if diedConn then diedConn:Disconnect() end
                            if timeoutTask then task.cancel(timeoutTask) end
                        end
                        stoppedConn = track.Stopped:Connect(cleanup)
                        endedConn = track.Ended:Connect(cleanup)
                        diedConn = hum.Died:Connect(cleanup)
                        local animDuration = (track.Length > 0 and track.Length) or 1.5
                        timeoutTask = task.delay(animDuration + 0.1, cleanup)
                    end
                end
            end
        end)
    end
    if p.Character then onChar(p.Character) end
    p.CharacterAdded:Connect(onChar)
end

    fl.BackgroundColor3 = Color3.fromRGB(0, 255, 100)
    Instance.new("UICorner", fl).CornerRadius = UDim.new(1, 0)
    local kn = Instance.new("Frame", tr)
    kn.Size = UDim2.fromOffset(11, 11)
    kn.Position = UDim2.new((value - minV) / (maxV - minV), -5, 0.5, -5)
    kn.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    Instance.new("UICorner", kn).CornerRadius = UDim.new(1, 0)
    local function setV(v)
        v = math.clamp(v, minV, maxV)
        if not isF then v = math.floor(v) end
        value = v
        local pct = (v - minV) / (maxV - minV)
        local d = isF and string.format("%.2f", v) or tostring(math.floor(v))
        lbl.Text = name .. ": " .. d
        fl.Size = UDim2.new(pct, 0, 1, 0)
        kn.Position = UDim2.new(pct, -5, 0.5, -5)
        pcall(cb, v)
    end
    local drag = false
    tr.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            drag = true
            local rx = input.Position.X - tr.AbsolutePosition.X
            setV(minV + math.clamp(rx / tr.AbsoluteSize.X, 0, 1) * (maxV - minV))
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if drag and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local rx = input.Position.X - tr.AbsolutePosition.X
            setV(minV + math.clamp(rx / tr.AbsoluteSize.X, 0, 1) * (maxV - minV))
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then 
            if drag then
                drag = false
                -- Gửi thông báo khi thả chuột (kết thúc kéo)
                if msgFormat then
                    local finalMsg = msgFormat:gsub("{val}", tostring(math.floor(value)))
                    Notify(finalMsg, true)
                else
                    Notify(name .. " ĐÃ CHỈNH: " .. tostring(math.floor(value)) .. " 🍢", true)
                end
            end
        end
    end)
end

local function Valid(p)
    if not p or p == LocalPlayer then return false end
    local c = p.Character
    if not c then return false end
    local r = c:FindFirstChild("HumanoidRootPart")
    local h = c:FindFirstChildOfClass("Humanoid")
    return r and h and h.Health > 0
end

local function FindNearest(range)
    local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not mr then return nil end
    local best, bd = nil, range or 100
    for _, p in ipairs(Players:GetPlayers()) do
        if Valid(p) then
            local r = p.Character:FindFirstChild("HumanoidRootPart")
            if r then
                local d = (mr.Position - r.Position).Magnitude
                if d < bd then bd = d; best = p end
            end
        end
    end
    return best
end

local function FindByName(name)
    if not name or name == "" then return nil end
    name = string.lower(name)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and string.find(string.lower(p.Name), name, 1, true) then
            return p
        end
    end
    return nil
end

local function GetTool()
    local c = LocalPlayer.Character
    if not c then return nil end
    local t = c:FindFirstChildOfClass("Tool")
    if t then return t end
    local bp = LocalPlayer:FindFirstChild("Backpack")
    if bp then
        for _, x in ipairs(bp:GetChildren()) do
            if x:IsA("Tool") then x.Parent = c; return x end
        end
    end
    return nil
end

task.spawn(function()

            pcall(function()

                    local name = Config.TargetName
                    if not name or name == "" then
                        name = TargetBox.Text
                        Config.TargetName = name
                    end
                    if name == "" then Config.Target = FindNearest(Config.ORRange)
                    else Config.Target = FindByName(name) end
                    if not Config.Target then
                        statusLbl.Text = "Cho '" .. name .. "' vao game..."
                        statusLbl.TextColor3 = Color3.fromRGB(255, 200, 100)
                        ORBuffer = {}; ORLastTarget = nil
                        return
                    else
                        statusLbl.Text = "Che do: " .. Config.ORMode .. " | " .. Config.Target.Name
                        statusLbl.TextColor3 = Color3.fromRGB(0, 255, 100)
                    end
                end
                if ORLastTarget ~= Config.Target then ORBuffer = {}; ORLastTarget = Config.Target end
                local tc = Config.Target.Character
                local tRoot = tc:FindFirstChild("HumanoidRootPart")
                local tHum = tc:FindFirstChildOfClass("Humanoid")
                local myC = LocalPlayer.Character
                local myRoot = myC and myC:FindFirstChild("HumanoidRootPart")
                local myHum = myC and myC:FindFirstChildOfClass("Humanoid")
                if not (tRoot and myRoot and myHum and tHum) then return end
                local dist = (myRoot.Position - tRoot.Position).Magnitude
                if dist > 2000 then return end

                local now = tick()
                table.insert(ORBuffer, {t=now, pos=tRoot.Position, cf=tRoot.CFrame})
                while #ORBuffer > 1 and (now - ORBuffer[1].t) > Config.ORDelay do
                    table.remove(ORBuffer, 1)
                end
                local dp, dcf = tRoot.Position, tRoot.CFrame
                if #ORBuffer > 0 then
                    local o = ORBuffer[1]
                    if now - o.t >= Config.ORDelay * 0.8 then dp = o.pos; dcf = o.cf end
                end
                local tp = dp
                local fp
                if Config.ORMode == "FOLLOW" then fp = tp - dcf.LookVector * Config.ORBehind
                elseif Config.ORMode == "STICK" then
                    local dir = tp - myRoot.Position
                    if dir.Magnitude > 0.1 then dir = dir.Unit else dir = Vector3.new(0,0,1) end
                    fp = tp - dir * Config.ORBehind
                elseif Config.ORMode == "BEHIND" then fp = tp - dcf.LookVector * Config.ORBehind
                elseif Config.ORMode == "ORBIT" then
                    Config.orbitAngle = Config.orbitAngle + Config.ORSpeed
                    fp = tp + Vector3.new(math.cos(Config.orbitAngle)*2, 0.5, math.sin(Config.orbitAngle)*2)
                else fp = tp - dcf.LookVector * Config.ORBehind end

                if State.LAGMode or Config.ORMode == "SLIDE" then
                    local toT = fp - myRoot.Position
                    local d = toT.Magnitude
                    if d > 1.5 then
                        local sp = Config.ORWalkSpeed
                        if d > 40 then sp = sp * 3 elseif d > 20 then sp = sp * 2 end
                        myRoot.AssemblyLinearVelocity = toT.Unit * sp
                        myRoot.AssemblyAngularVelocity = Vector3.zero
                        pcall(function()
                            myRoot.CFrame = CFrame.lookAt(myRoot.Position, tRoot.Position)
                        end)
                    else
                        myRoot.AssemblyLinearVelocity = Vector3.zero
                    end
                else
                    pcall(function()
                        myRoot.CFrame = CFrame.lookAt(fp + Vector3.new(0, 2.5, 0), tRoot.Position + Vector3.new(0, 2.5, 0))
                        myRoot.AssemblyLinearVelocity = Vector3.zero
                        myRoot.AssemblyAngularVelocity = Vector3.zero
                    end)
                end
                if dist <= Config.ORNear then HitAll() end
            end)
        end
    end
end)

RunService.RenderStepped:Connect(function()
    if not isEnabled then return end
    
    if activeM2Count > 0 then
        rawUnblock()
        return
    end
    
    local myChar = LocalPlayer.Character
    if not myChar then return end
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end
    local myPos = myRoot.Position
    local m1Detected = false
    
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local cRoot = p.Character:FindFirstChild("HumanoidRootPart")
            local cHum = p.Character:FindFirstChild("Humanoid")
            if cRoot and cHum and cHum.Health > 0 then
                local dx = cRoot.Position.X - myPos.X
                local dy = cRoot.Position.Y - myPos.Y
                local dz = cRoot.Position.Z - myPos.Z
                
                if (dx * dx + dy * dy + dz * dz) <= COMBAT_RANGE_SQ then
                    local anims = cHum:GetPlayingAnimationTracks()
                    for i = 1, #anims do
                        local track = anims[i]
                        local animObj = track.Animation
                        if animObj and M1_Hash[animObj.AnimationId] then
                            m1Detected = true
                            break
                        end
                    end
                end
            end
        end
        if m1Detected then break end
    end
    
    if m1Detected then
        lastM1Time = tick()
        rawBlock()
    else
        if isBlocking and (tick() - lastM1Time) > BUFFER_TIME then
            rawUnblock()
        end
    end
end)
